% Preamble additions for the manuscript build.
%
% The tables carry long descriptive cells and several are wider than a printed
% line at the body size, so every table is set one step smaller. This is enough
% for all but one of them and leaves the body text unchanged.
%
% Long words in a language that builds them by suffix break badly at the default
% penalties, so the tolerance for stretched lines is raised and hyphenation is
% allowed to work harder before a line is left overfull.

\usepackage{etoolbox}
\AtBeginEnvironment{longtable}{\small}
\AtBeginEnvironment{tabular}{\small}

\tolerance=2000
\emergencystretch=2em
\hbadness=4000
